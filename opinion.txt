/* ==============================================================
   VARIABLES
   ============================================================== */


let valoracion = 0;

let seguirLeyendo = "";


const estrellas =
  document.querySelectorAll(
    ".estrella"
  );


const opcionesSeguir =
  document.querySelectorAll(
    ".opcion"
  );


const aviso =
  document.getElementById(
    "aviso"
  );


const emailContenedor =
  document.getElementById(
    "emailContenedor"
  );


const email =
  document.getElementById(
    "email"
  );


const formulario =
  document.getElementById(
    "formulario"
  );


const botonEnviar =
  document.getElementById(
    "botonEnviar"
  );


const mensajeError =
  document.getElementById(
    "error"
  );



/* ==============================================================
   ESTRELLAS
   ============================================================== */


estrellas.forEach(
  function(estrella){


    estrella.addEventListener(
      "click",
      function(){


        valoracion =
          Number(
            estrella.dataset.valor
          );


        estrellas.forEach(
          function(e){


            const valor =
              Number(
                e.dataset.valor
              );


            e.classList.toggle(
              "seleccionada",
              valor <= valoracion
            );


          }
        );


        mensajeError.textContent =
          "";


      }
    );


  }
);



/* ==============================================================
   SEGUIR LEYENDO
   ============================================================== */


opcionesSeguir.forEach(
  function(opcion){


    opcion.addEventListener(
      "click",
      function(){


        seguirLeyendo =
          opcion.dataset.seguir;


        opcionesSeguir.forEach(
          function(o){


            o.classList.toggle(
              "seleccionada",
              o === opcion
            );


          }
        );


        mensajeError.textContent =
          "";


      }
    );


  }
);



/* ==============================================================
   AVISO DE LANZAMIENTO
   ============================================================== */


aviso.addEventListener(
  "change",
  function(){


    emailContenedor
      .classList
      .toggle(
        "visible",
        aviso.checked
      );


    if(
      aviso.checked
    ){

      setTimeout(
        function(){

          email.focus();

        },
        100
      );

    }


  }
);



/* ==============================================================
   ENVIAR OPINIÓN
   ============================================================== */


formulario.addEventListener(
  "submit",
  async function(event){


    event.preventDefault();


    mensajeError.textContent =
      "";


    /* VALIDAR VALORACIÓN */


    if(
      valoracion < 1 ||
      valoracion > 5
    ){

      mensajeError.textContent =
        CONFIG.errorValoracion;

      return;

    }



    /* VALIDAR SEGUIR LEYENDO */


    if(
      seguirLeyendo !== "Sí" &&
      seguirLeyendo !== "No"
    ){

      mensajeError.textContent =
        CONFIG.errorSeguir;

      return;

    }



    /* VALIDAR EMAIL SI QUIERE AVISO */


    const emailValor =
      email.value.trim();


    if(
      aviso.checked
    ){


      const patronEmail =
        /^[^\s@]+@[^\s@]+\.[^\s@]+$/;


      if(
        !patronEmail.test(
          emailValor
        )
      ){

        mensajeError.textContent =
          CONFIG.errorEmail;

        return;

      }

    }



    /* BLOQUEAR BOTÓN */


    botonEnviar.disabled =
      true;


    botonEnviar.textContent =
      CONFIG.botonEnviando;



    /* CREAR DATOS */


    const datos = {

      tipo:
        "opinion",

      valoracion:
        valoracion,

      comentario:
        document
          .getElementById(
            "comentario"
          )
          .value
          .trim(),

      seguirLeyendo:
        seguirLeyendo,

      email:
        aviso.checked
          ? emailValor
          : "",

      avisoLanzamiento:
        aviso.checked,

      usuarioRRSS:
        document
          .getElementById(
            "usuarioRRSS"
          )
          .value
          .trim()

    };



    /* ENVIAR */


    try{


      await fetch(
        CONFIG.url,
        {

          method:
            "POST",

          mode:
            "no-cors",

          headers:{

            "Content-Type":
              "text/plain;charset=utf-8"

          },

          body:
            JSON.stringify(
              datos
            )

        }
      );



      /* MOSTRAR GRACIAS */


      document
        .getElementById(
          "encuesta"
        )
        .style.display =
        "none";


      document
        .getElementById(
          "gracias"
        )
        .classList
        .add(
          "visible"
        );


      window.scrollTo({
        top:0,
        behavior:"smooth"
      });


    }


    catch(error){


      botonEnviar.disabled =
        false;


      botonEnviar.textContent =
        CONFIG.boton;


      mensajeError.textContent =
        CONFIG.errorEnvio;


    }


  }
);
